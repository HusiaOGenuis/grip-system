    <script>
        let SESSION_TOKEN = null;

        async function safeApiCall(url, options) {
            try {
                const res = await fetch(url, options);
                if (res.status === 401) {
                    return { error: true, message: "Session expired or authentication invalid." };
                }
                const text = await res.text();
                return text ? JSON.parse(text) : { error: true, message: "Empty payload received." };
            } catch (e) {
                console.error("Network or parse trace error:", e);
                return { error: true, message: "Local connection routing failure." };
            }
        }

        async function executeLogin() {
            const email = document.getElementById("email").value;
            const password = document.getElementById("password").value;
            const statusBox = document.getElementById("status");

            if (!email || !password) {
                statusBox.className = "error";
                statusBox.innerText = "Please complete all fields.";
                return;
            }

            statusBox.className = "";
            statusBox.innerText = "Authenticating with security token gate...";

            const data = await safeApiCall("/api/login", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ email, password })
            });

            if (data.error) {
                statusBox.className = "error";
                statusBox.innerText = data.message;
                return;
            }

            SESSION_TOKEN = data.access_token;
            statusBox.className = "success";
            statusBox.innerText = "Identity verified. Session Token generated successfully.";
        }

        async function runDecisionEngine() {
            const scoreVal = document.getElementById("score").value;
            const resultElement = document.getElementById("result");

            if (!SESSION_TOKEN) {
                resultElement.innerText = JSON.stringify({ error: true, message: "Unauthorized Request: Please authenticate first." }, null, 2);
                return;
            }

            const data = await safeApiCall("/api/decision", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    "Authorization": "Bearer " + SESSION_TOKEN
                },
                body: JSON.stringify({ score: scoreVal })
            });

            resultElement.innerText = JSON.stringify(data, null, 2);
        }
    </script>
