<script>
/* =========================================
GLOBAL SESSION STORAGE
========================================= */
let GRIP_TOKEN = null;
/* =========================================
PANEL NAVIGATION
========================================= */
function switchPanel(panelId){
document
.querySelectorAll('.panel')
.forEach(panel => {
panel.classList.remove('active');
});
document
.getElementById(panelId)
.classList.add('active');
}
/* =========================================
LOGIN
========================================= */
document
.getElementById('loginBtn')
.addEventListener('click', async function(){
const email =
document.getElementById('email').value.trim();
const password =
document.getElementById('password').value.trim();
const statusBox =
document.getElementById('statusBox');
if(!email || !password){
statusBox.innerText =
'Please complete all fields.';
return;
}
statusBox.innerText =
'Authenticating...';
try{
const response = await fetch(
'https://grip-dashboard.onrender.com/login',
{
method:'POST',
headers:{
'Content-Type':'application/json'
},
body:JSON.stringify({
email:email,
password:password
})
}
);
const data = await response.json();
if(!response.ok){
throw new Error(
data.detail || 'Authentication failed.'
);
}
GRIP_TOKEN = data.access_token;
statusBox.innerText =
'Authentication successful.';
document
.getElementById('loginWrapper')
.style.display = 'none';
document
.getElementById('app')
.style.display = 'block';
}
catch(err){
console.error(err);
statusBox.innerText =
err.message;
}
});
/* =========================================
GOVERNANCE ENGINE
========================================= */
async function executeEvaluation(){
const resultBox =
document.getElementById('resultBox');
const score =
document.getElementById('scoreInput').value;
if(!score){
resultBox.innerText =
'Please enter a valid score.';
return;
}
resultBox.innerText =
'Executing governance analysis...';
try{
const response = await fetch(
'https://YOUR-RENDER-APP.onrender.com/analyze',
{
method:'POST',
headers:{
'Content-Type':'application/json',
'Authorization':
'Bearer ' + GRIP_TOKEN
},
body:JSON.stringify({
score:parseInt(score)
})
}
);
const data = await response.json();
if(!response.ok){
throw new Error(
data.detail || 'Analysis failed.'
);
}
resultBox.innerText =
'Verdict: ' + data.verdict +
'\nReason: ' + data.rationale;
}
catch(err){
console.error(err);
resultBox.innerText =
err.message;
}
}
</script>
